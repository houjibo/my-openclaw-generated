```typescript
// 自动保存（防抖）
useEffect(() => {
  if (!isDirty || isSaving) return;
  const timer = setTimeout(() => {
    handleSave();
  }, 2000);
  return () => clearTimeout(timer);
}, [isDirty, isSaving, handleSave]);

// 表单变化检测
useEffect(() => {
  const hasChanges = 
    avatarUrl !== user.avatar ||
    name !== user.name ||
    bio !== user.bio;
  setIsDirty(hasChanges);
}, [avatarUrl, name, bio, user.avatar, user.name, user.bio]);
```
